import { cloneMockAccount } from './mock-data.js';

const STORAGE_PREFIX = 'genshin_sheet_mock_v1_';
const SESSION_TOKEN_KEY = 'genshin_sheet_token_v1';

function getConfig() {
  return window.GENSHIN_SHEET_CONFIG || { APPS_SCRIPT_URL: '' };
}

export function getSessionToken() {
  try {
    return sessionStorage.getItem(SESSION_TOKEN_KEY) || '';
  } catch {
    return '';
  }
}

function setSessionToken(token) {
  try {
    if (token) {
      sessionStorage.setItem(SESSION_TOKEN_KEY, token);
    } else {
      sessionStorage.removeItem(SESSION_TOKEN_KEY);
    }
  } catch (err) {
    console.warn('sessionStorage token write failed:', err);
  }
}

export function clearSessionToken() {
  setSessionToken('');
}

function requireSessionToken() {
  const token = getSessionToken();
  if (!token) {
    throw new Error('Session expired. Please sign in again.');
  }
  return token;
}

async function getAppsScript(params, errorLabel) {
  const cfg = getConfig();
  const url = new URL(cfg.APPS_SCRIPT_URL);
  for (const [k, v] of Object.entries(params)) {
    if (v !== undefined && v !== null) {
      url.searchParams.set(k, String(v));
    }
  }
  const res = await fetch(url.toString(), { method: 'GET' });
  if (!res.ok) {
    throw new Error(`${errorLabel} (${res.status})`);
  }
  const json = await res.json();
  if (json.error) throw new Error(json.error);
  return json;
}

async function postAppsScript(payload, errorLabel) {
  const cfg = getConfig();
  const res = await fetch(cfg.APPS_SCRIPT_URL, {
    method: 'POST',
    headers: { 'Content-Type': 'text/plain;charset=utf-8' },
    body: JSON.stringify(payload)
  });
  if (!res.ok) {
    throw new Error(`${errorLabel} (${res.status})`);
  }
  const json = await res.json();
  if (json.error) throw new Error(json.error);
  return json;
}

export function isOfflineMode() {
  const cfg = getConfig();
  return !cfg.APPS_SCRIPT_URL || cfg.APPS_SCRIPT_URL.trim() === '';
}

function loadLocalMock(username) {
  try {
    const raw = localStorage.getItem(`${STORAGE_PREFIX}${username}`);
    if (raw) {
      return JSON.parse(raw);
    }
  } catch (err) {
    console.warn('localStorage read failed:', err);
  }
  const fresh = cloneMockAccount(username);
  saveLocalMock(username, fresh);
  return fresh;
}

export function saveLocalMock(username, data) {
  try {
    localStorage.setItem(`${STORAGE_PREFIX}${username}`, JSON.stringify(data));
  } catch (err) {
    console.warn('localStorage write failed:', err);
  }
}

function normalizeSheetRowCharacter(r) {
  const level = Number(r.level) || 1;
  return {
    key: r.key || r.char_key,
    owned: r.owned === undefined ? true : Boolean(r.owned),
    level,
    constellation: Number(r.constellation) || 0,
    ascension: Number(r.ascension) || 0,
    talent: r.talent || {
      auto: Number(r.talent_auto) || 1,
      skill: Number(r.talent_skill) || 1,
      burst: Number(r.talent_burst) || 1
    },
    targetLevel: Number(r.targetLevel ?? r.target_lvl) || 90,
    targetTalents: r.targetTalents || r.target_talents || '9/9/9',
    status: r.status || (level >= 90 ? 'Built' : level >= 70 ? 'In Progress' : 'Bench'),
    priorityRank: r.priorityRank ?? r.priority_rank ?? '',
    swapTag: r.swapTag || '',
    notes: r.notes || ''
  };
}

function normalizeAccount(raw) {
  if (!raw || typeof raw !== 'object') return raw;

  const rosterRows = Array.isArray(raw.roster) ? raw.roster.map(normalizeSheetRowCharacter).filter((c) => c.key) : [];
  const sectionChars = Array.isArray(raw.characters) ? raw.characters.map(normalizeSheetRowCharacter).filter((c) => c.key) : [];

  let characters = sectionChars;
  if (sectionChars.length === 0 && rosterRows.length > 0) {
    characters = rosterRows;
  } else if (rosterRows.length > 0 && sectionChars.length > 0) {
    const byKey = new Map(sectionChars.map((c) => [c.key, c]));
    for (const r of rosterRows) {
      const prev = byKey.get(r.key);
      byKey.set(r.key, prev ? { ...r, status: prev.status || r.status, swapTag: prev.swapTag || r.swapTag } : r);
    }
    characters = Array.from(byKey.values());
  }

  const weapons = (Array.isArray(raw.weapons) ? raw.weapons : [])
    .map((w, idx) => ({
      id: w.id || w.weapon_uid || `w-${idx + 1}`,
      key: w.key || w.weapon_key,
      level: Number(w.level) || 1,
      ascension: Number(w.ascension) || 0,
      refinement: Number(w.refinement) || 1,
      location: w.location || '',
      lock: Boolean(w.lock)
    }))
    .filter((w) => Boolean(w.key));

  return {
    ...raw,
    characters,
    weapons
  };
}

/**
 * Authenticates user via POST (action: "login") so passphrase stays out of URLs,
 * stores the returned temporary session token, and returns the account payload.
 */
export async function loginAccount(username, passphrase = '') {
  const cfg = getConfig();
  const normalized = (username || '').trim().toLowerCase();

  if (!cfg.ACCOUNTS?.[normalized]) {
    throw new Error('Invalid account key or passphrase.');
  }

  if (isOfflineMode()) {
    return {
      ok: true,
      mode: 'offline',
      account: loadLocalMock(normalized)
    };
  }

  if (!passphrase) {
    throw new Error('Passphrase is required in live mode.');
  }

  const json = await postAppsScript(
    {
      action: 'login',
      user: normalized,
      passphrase
    },
    'Apps Script login failed'
  );

  const sessionToken = json.sessionToken || json.token;
  if (!sessionToken) {
    throw new Error('Login succeeded but server did not issue a session token.');
  }
  setSessionToken(sessionToken);

  return {
    ok: true,
    mode: 'live',
    account: normalizeAccount(json.account || json.data) || loadLocalMock(normalized)
  };
}

/**
 * Restores an existing session on page reload using the temporary session token.
 */
export async function restoreAccountSession(username) {
  const cfg = getConfig();
  const normalized = (username || '').trim().toLowerCase();

  if (!cfg.ACCOUNTS?.[normalized]) {
    throw new Error('Unknown account.');
  }

  if (isOfflineMode()) {
    return {
      ok: true,
      mode: 'offline',
      account: loadLocalMock(normalized)
    };
  }

  const token = requireSessionToken();
  const json = await getAppsScript(
    {
      action: 'getSheet',
      user: normalized,
      token
    },
    'Session restore failed'
  );

  return {
    ok: true,
    mode: 'live',
    account: normalizeAccount(json.account || json.data) || loadLocalMock(normalized)
  };
}

/**
 * Clears the local session token and best-effort revokes it on the backend.
 */
export async function logoutAccount() {
  const token = getSessionToken();
  clearSessionToken();

  if (!isOfflineMode() && token) {
    try {
      await postAppsScript({ action: 'logout', token }, 'Logout failed');
    } catch {
      // Best-effort revocation; local token is already cleared.
    }
  }
}

/**
 * Fetches partner account data in read-only mode.
 */
export async function fetchPartnerAccount(partnerUsername) {
  if (isOfflineMode()) {
    return loadLocalMock(partnerUsername);
  }

  const token = requireSessionToken();
  const json = await getAppsScript(
    {
      action: 'getSheet',
      user: partnerUsername,
      token
    },
    'Failed to fetch partner sheet'
  );
  return normalizeAccount(json.account || json.data);
}

/**
 * Sends a single upsertRow POST with Content-Type: text/plain to {user}_Inventory (or section).
 */
export async function upsertInventoryRow(username, key, rowPayload) {
  if (isOfflineMode()) {
    const current = loadLocalMock(username);
    current.inventory = current.inventory || {};
    current.inventory[key] = {
      ...(current.inventory[key] || {}),
      ...rowPayload
    };
    saveLocalMock(username, current);
    return { ok: true, mode: 'offline', key, savedAt: Date.now() };
  }

  const token = requireSessionToken();
  await postAppsScript(
    {
      action: 'upsertRow',
      token,
      user: username,
      sheet: `${username}_Inventory`,
      key,
      data: rowPayload
    },
    'upsertRow failed'
  );
  return { ok: true, mode: 'live', key, savedAt: Date.now() };
}

/**
 * Sends a single upsertRow POST with Content-Type: text/plain to {user}_Roster.
 */
export async function upsertRosterRow(username, charKey, rowPayload) {
  if (isOfflineMode()) {
    const current = loadLocalMock(username);
    current.characters = current.characters || [];
    const idx = current.characters.findIndex((c) => c.key === charKey);
    const updated = {
      ...(idx >= 0 ? current.characters[idx] : { key: charKey }),
      ...rowPayload
    };
    if (idx >= 0) current.characters[idx] = updated;
    else current.characters.push(updated);
    saveLocalMock(username, current);
    return { ok: true, mode: 'offline', key: charKey, savedAt: Date.now() };
  }

  const token = requireSessionToken();
  await postAppsScript(
    {
      action: 'upsertRow',
      token,
      user: username,
      sheet: `${username}_Roster`,
      key: charKey,
      data: rowPayload
    },
    'upsertRow failed'
  );
  return { ok: true, mode: 'live', key: charKey, savedAt: Date.now() };
}

/**
 * Persists non-inventory account sections (wallet, checklist, prefarmQueue, characters, teams).
 */
export async function saveAccountSection(username, section, payload) {
  if (isOfflineMode()) {
    const current = loadLocalMock(username);
    current[section] = payload;
    saveLocalMock(username, current);
    return { ok: true, mode: 'offline' };
  }

  const token = requireSessionToken();
  return postAppsScript(
    {
      action: 'upsertSection',
      token,
      user: username,
      section,
      data: payload
    },
    'saveAccountSection failed'
  );
}

/**
 * Sends a single POST (action: "importGoodJson", Content-Type: text/plain) to Apps Script.
 */
export async function postImportGoodJson(username, goodPayload, mergedAccountData) {
  if (isOfflineMode()) {
    saveLocalMock(username, mergedAccountData);
    return { ok: true, mode: 'offline', syncedAt: Date.now() };
  }

  const token = requireSessionToken();
  return postAppsScript(
    {
      action: 'importGoodJson',
      token,
      user: username,
      good: goodPayload
    },
    'importGoodJson failed'
  );
}

function showToast(message, tone = 'info') {
  const toast = document.getElementById('toast');
  if (!toast) return;
  toast.textContent = message;
  toast.className = `toast is-visible tone-${tone}`;
  clearTimeout(showToast._timer);
  showToast._timer = setTimeout(() => {
    toast.className = 'toast';
  }, 3200);
}

function renderActiveTab(options = {}) {
  const account = getActiveAccount();
  if (!account) return;

  // Strictly show active tab only
  document.querySelectorAll('.tab-view').forEach((sec) => {
    const isActive = sec.dataset.tab === state.activeTab;
    sec.hidden = !isActive;
    sec.classList.toggle('active', isActive);
  });

  const viewState = {
    account,
    isReadOnly: state.viewingPartner,
    serverId: state.serverId,
    hudDayOverride: state.hudDayOverride,
    hudShowAllDomains: state.hudShowAllDomains,
    vaultFilter: state.vaultFilter,
    rosterFilter: state.rosterFilter
  };

  const container = document.querySelector(`.tab-view[data-tab="${state.activeTab}"]`);
  if (!container) return;

  if (state.activeTab === 'hud') {
    renderHudTab(container, viewState, {
      onChangeHudDay: (day) => {
        state.hudDayOverride = day;
        renderActiveTab();
      },
      onToggleAllDomains: () => {
        state.hudShowAllDomains = !state.hudShowAllDomains;
        renderActiveTab();
      },
      onUpdateWallet: async (field, val) => {
        state.ownAccount.wallet = { ...(state.ownAccount.wallet || {}), [field]: val };
        await saveAccountSection(state.sessionUser, 'wallet', state.ownAccount.wallet);
        renderActiveTab();
        showToast(`Updated ${field}`, 'mint');
      },
      onUpdateWalletPatch: async (patch, toastMsg = 'Updated Resin') => {
        state.ownAccount.wallet = { ...(state.ownAccount.wallet || {}), ...patch };
        await saveAccountSection(state.sessionUser, 'wallet', state.ownAccount.wallet);
        renderActiveTab();
        showToast(toastMsg, 'mint');
      },
      onToggleChecklist: async (group, id, done) => {
        const list = state.ownAccount.checklist?.[group] || [];
        const item = list.find((i) => i.id === id);
        if (item) item.done = done;
        await saveAccountSection(state.sessionUser, 'checklist', state.ownAccount.checklist);
        renderActiveTab();
      },
      onOpenServerModal: () => openServerModal()
    });
  } else if (state.activeTab === 'vault') {
    renderVaultTab(container, viewState, {
      onChangeVaultFilter: (patch, opts) => {
        state.vaultFilter = { ...state.vaultFilter, ...patch };
        renderActiveTab(opts);
      },
      onInventoryRowSaved: (key, payload) => {
        state.ownAccount.inventory[key] = payload;
        if (key === 'Mora') {
          state.ownAccount.wallet.mora = payload.count;
        }
      }
    });
  } else if (state.activeTab === 'prefarm') {
    renderPrefarmTab(container, viewState, {
      onAddPrefarmTarget: async (target) => {
        const queue = state.ownAccount.prefarmQueue || [];
        const charName = getCharacterMeta(target.characterKey).name || target.characterKey;
        if (queue.some((t) => t.characterKey === target.characterKey)) {
          showToast(`You've already added ${charName} to the Prefarm Queue.`, 'coral');
          return;
        }
        const uid = `pf-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`;
        const tier = Math.max(1, Number(target.tier) || 1);
        const nextQueue = sortAndLabelPrefarmQueue([
          ...queue,
          { id: uid, rank: queue.length + 1, ...target, tier }
        ]);
        state.ownAccount.prefarmQueue = nextQueue;
        await saveAccountSection(state.sessionUser, 'prefarmQueue', nextQueue);
        renderActiveTab();
        showToast(`Added ${charName} to Tier #${tier}`, 'mint');
      },
      onMovePrefarmTarget: async (id, dir) => {
        const queue = sortAndLabelPrefarmQueue(state.ownAccount.prefarmQueue || []);
        const idx = queue.findIndex((t) => String(t.id) === String(id));
        if (idx === -1) return;
        const swapIdx = dir === 'up' ? idx - 1 : idx + 1;
        if (swapIdx < 0 || swapIdx >= queue.length) return;
        if (queue[idx].tier !== queue[swapIdx].tier) return;
        [queue[idx], queue[swapIdx]] = [queue[swapIdx], queue[idx]];
        queue.forEach((t, i) => (t.rank = i + 1));
        const normalized = sortAndLabelPrefarmQueue(queue);
        state.ownAccount.prefarmQueue = normalized;
        renderActiveTab();
        await saveAccountSection(state.sessionUser, 'prefarmQueue', normalized);
      },
      onRemovePrefarmTarget: async (id, fallbackRank) => {
        const prev = sortAndLabelPrefarmQueue(state.ownAccount.prefarmQueue || []);
        const queue = sortAndLabelPrefarmQueue(
          prev
            .filter((t, i) => {
              if (t.id !== undefined && t.id !== null && String(t.id) !== '') {
                return String(t.id) !== String(id);
              }
              return Number(t.rank ?? i + 1) !== Number(fallbackRank);
            })
            .map((t, i) => ({ ...t, id: t.id || `pf-${Date.now()}-${i}` }))
        );
        state.ownAccount.prefarmQueue = queue;
        renderActiveTab();
        showToast('Removed target from Prefarm Queue', 'mint');
        try {
          await saveAccountSection(state.sessionUser, 'prefarmQueue', queue);
        } catch (err) {
          showToast(err?.message || 'Failed to sync removal', 'coral');
        }
      },
      onEditPrefarmTarget: async (id, field, val) => {
        const target = (state.ownAccount.prefarmQueue || []).find((t) => String(t.id) === String(id));
        if (!target) return;
        if (field === 'tier') target.tier = Math.min(10, Math.max(1, val));
        else if (field === 'currentLevel') target.currentLevel = Math.min(95, Math.max(1, val));
        else if (field === 'targetLevel') target.targetLevel = Math.min(95, Math.max(1, val));
        else if (field === 'tAuto') target.targetTalent.auto = Math.min(10, Math.max(1, val));
        else if (field === 'tSkill') target.targetTalent.skill = Math.min(10, Math.max(1, val));
        else if (field === 'tBurst') target.targetTalent.burst = Math.min(10, Math.max(1, val));
        else if (field === 'weaponKey') {
          target.weaponKey = val || '';
          if (target.weaponKey && !target.weaponTargetLevel) {
            target.weaponCurrentLevel = 1;
            target.weaponTargetLevel = 90;
          }
        } else if (field === 'weaponCurrentLevel') target.weaponCurrentLevel = Math.min(90, Math.max(1, val));
        else if (field === 'weaponTargetLevel') target.weaponTargetLevel = Math.min(90, Math.max(1, val));
        state.ownAccount.prefarmQueue = sortAndLabelPrefarmQueue(state.ownAccount.prefarmQueue || []);
        renderActiveTab();
        await saveAccountSection(state.sessionUser, 'prefarmQueue', state.ownAccount.prefarmQueue);
      }
    });
  } else if (state.activeTab === 'roster') {
    renderRosterTab(container, viewState, {
      onChangeRosterFilter: (patch, opts) => {
        state.rosterFilter = { ...state.rosterFilter, ...patch };
        if (!opts?.silent) renderActiveTab(opts);

        if (!team || !team.members[memberIdx]) return;

      },

        if (!team) return;
        team.testLab = { ...(team.testLab || {}), [field]: val };
        await saveAccountSection(state.sessionUser, 'teams', state.ownAccount.teams);
        showToast('Saved Test Lab note', 'mint');

  }

  if (options?.preserveFocusSelector) {

  }

      updateTopBar();
      renderActiveTab();

      showToast(err.message, 'coral');
    }
  });

  document.getElementById('btn-logout')?.addEventListener('click', () => {
    logoutAccount();
    sessionStorage.removeItem(SESSION_KEY);
    state.sessionUser = null;
    state.ownAccount = null;
    state.partnerAccount = null;
    state.viewingPartner = false;
    switchToLoginView();
  });

  setupGoodImportModal();
  setupServerModal();
  setupSystemDrawer();

  // Restore session from sessionStorage
  try {
    const saved = sessionStorage.getItem(SESSION_KEY);

      const parsed = JSON.parse(saved);
      if (parsed?.user) {
        const res = await restoreAccountSession(parsed.user);

      }
    }
  } catch (err) {

import { getCatalog, getLiveAssetEntry } from './catalog.js?v=20261113j';

/**
 * Hybrid Icon Resolver
 * First checks /assets/custom/{type}/{pascalKey}.png (custom Aseprite PNGs),
 * then falls back to official Genshin icon URLs mapped from data/catalog.json.
 *
 * @param {'characters'|'weapons'|'materials'} type
 * @param {string} pascalKey
 * @returns {string}
 */
export function getAssetUrl(type, pascalKey) {
  const safeType = encodeURIComponent(type || 'materials');
  const safeKey = encodeURIComponent(pascalKey || 'Unknown');
  return `./assets/custom/${safeType}/${safeKey}.png`;
}

/**
 * Looks up the official Genshin icon URL from data/catalog.json.
 */
export function getFallbackAssetUrl(type, pascalKey) {
  const catalog = getCatalog();
  if (!pascalKey) return '';
  const live = getLiveAssetEntry(type, pascalKey);
  if (live?.icon) return live.icon;

  const bucket = catalog?.[type] || {};
  if (bucket[pascalKey]?.icon) return bucket[pascalKey].icon;

  // Handle Traveler variants (TravelerCryo -> Traveler or TravelerAnemo)
  if (type === 'characters' && pascalKey.startsWith('Traveler')) {
    return bucket.TravelerCryo?.icon || 'https://enka.network/ui/UI_AvatarIcon_PlayerGirl.png';
  }
  return '';
}

/**
 * Renders an <img> element string with hybrid custom -> catalog -> monogram fallback.
 */
export function renderAssetIcon(type, pascalKey, alt = '', className = 'asset-icon') {
  const customSrc = getAssetUrl(type, pascalKey);
  const fallbackSrc = getFallbackAssetUrl(type, pascalKey);
  const primarySrc = fallbackSrc || customSrc;
  const initials = (pascalKey || '?').slice(0, 2).toUpperCase();
  const safeAlt = (alt || pascalKey || '').replace(/"/g, '&quot;');

  return `<span class="asset-wrap ${className}" data-key="${safeAlt}">
    <img
      src="${primarySrc}"
      alt="${safeAlt}"
      class="asset-img${fallbackSrc ? '' : ' is-custom-pixel'}"
      referrerpolicy="no-referrer"
      onerror="this.style.display='none';this.nextElementSibling.style.display='flex';"
    />
    <span class="asset-monogram" style="display:none;">${initials}</span>
  </span>`;
}

/**
 * Effective Hero's Wit = 4★ + 0.25 × 3★ + 0.05 × 2★
 */
export const SERVER_REGIONS = {
  NA: { id: 'NA', label: 'America (NA)', short: 'NA', offsetHours: -5, resetLabel: '04:00 EST / 09:00 UTC' },
  EU: { id: 'EU', label: 'Europe (EU)', short: 'EU', offsetHours: 1, resetLabel: '04:00 CET / 03:00 UTC' },
  ASIA: { id: 'ASIA', label: 'Asia', short: 'ASIA', offsetHours: 8, resetLabel: '04:00 CST / 20:00 UTC' },
  SAR: { id: 'SAR', label: 'TW, HK, MO', short: 'TW/HK/MO', offsetHours: 8, resetLabel: '04:00 CST / 20:00 UTC' }
};

const WEEKDAY_NAMES = ['Sunday', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday'];

/**
 * Computes the current in-game day (adjusted for 04:00 server reset) and
 * time remaining until the next daily reset for the given server region.
 */
export function getServerResetInfo(serverId = 'NA', nowMs = Date.now()) {
  const server = SERVER_REGIONS[serverId] || SERVER_REGIONS.NA;
  const offsetMs = server.offsetHours * 3600_000;
  const resetBoundaryMs = 4 * 3600_000; // 04:00 server local time
  const inGameMs = nowMs + offsetMs - resetBoundaryMs;
  const inGameDate = new Date(inGameMs);
  const inGameDay = WEEKDAY_NAMES[inGameDate.getUTCDay()];

  const dayMs = 86400_000;
  const elapsedInDay = ((inGameMs % dayMs) + dayMs) % dayMs;
  const msUntilReset = dayMs - elapsedInDay;
  const totalMinutes = Math.floor(msUntilReset / 60_000);
  const hours = Math.floor(totalMinutes / 60);
  const minutes = totalMinutes % 60;

  return {
    server,
    inGameDay,
    msUntilReset,
    hours,
    minutes,
    countdownText: `${String(hours).padStart(2, '0')}h ${String(minutes).padStart(2, '0')}m`
  };
}

/**
 * Computes a weighted 0-100% readiness score from a target's requirement rows.
 */
export function calcTargetReadinessPct(reqRows = []) {
  if (!reqRows.length) return 100;
  let sumRatio = 0;
  for (const r of reqRows) {
    const need = Number(r.needed) || 0;
    if (need <= 0) {
      sumRatio += 1;
    } else {
      sumRatio += Math.min(1, Math.max(0, (Number(r.owned) || 0) / need));
    }
  }
  return Math.round((sumRatio / reqRows.length) * 100);
}

export function calcEffectiveHerosWit(inventory = {}) {
  const wits4 = Number(inventory.HerosWit?.count ?? 0);
  const exp3 = Number(inventory.AdventurersExperience?.count ?? 0);
  const adv2 = Number(inventory.WanderersAdvice?.count ?? 0);
  return {
    wits4,
    exp3,
    adv2,
    effectiveTotal: Number((wits4 + 0.25 * exp3 + 0.05 * adv2).toFixed(2))
  };
}

/**
 * Coin Purse: Primogems + Fates -> total pulls
 */
export function calcCoinPurse(wallet = {}) {
  const primogems = Number(wallet.primogems ?? 0);
  const intertwinedFates = Number(wallet.intertwinedFates ?? 0);
  const acquaintFates = Number(wallet.acquaintFates ?? 0);
  const primoPulls = Math.floor(primogems / 160);
  const limitedPulls = primoPulls + intertwinedFates;
  const totalPulls = limitedPulls + acquaintFates;
  return {
    primogems,
    intertwinedFates,
    acquaintFates,
    primoPulls,
    limitedPulls,
    totalPulls
  };
}

export function calcCompletionGauges(account) {
  const chars = account?.characters || [];
  const weapons = account?.weapons || [];

  const ownedChars = chars.filter((c) => c.status !== 'Unowned');
  const builtChars = ownedChars.filter((c) => c.status === 'Built' || (c.level >= 90 && (c.talent?.skill >= 8 || c.talent?.burst >= 8 || c.talent?.auto >= 8)));

  const ownedWeapons = weapons.length;
  const maxWeapons = weapons.filter((w) => Number(w.level) >= 90).length;

  return {
    builtChars: builtChars.length,
    ownedChars: ownedChars.length,
    charPct: ownedChars.length ? Math.round((builtChars.length / ownedChars.length) * 100) : 0,
    maxWeapons,
    ownedWeapons,
    weaponPct: ownedWeapons ? Math.round((maxWeapons / ownedWeapons) * 100) : 0
  };
}

function expWitsForLevel(level, curves) {
  const table = curves?.characterExpByLevelCap || {
    20: { mora: 24200, wits: 6 },
    40: { mora: 115800, wits: 29 },
    50: { mora: 232000, wits: 58 },
    60: { mora: 380000, wits: 95 },
    70: { mora: 590000, wits: 148 },
    80: { mora: 912000, wits: 228 },
    90: { mora: 1673400, wits: 419 },
    95: { mora: 2250000, wits: 565 }
  };
  const caps = Object.keys(table).map(Number).sort((a, b) => a - b);
  if (level <= 1) return { mora: 0, wits: 0 };
  for (let i = 0; i < caps.length; i++) {
    const cap = caps[i];
    if (level <= cap) {
      const prevCap = i === 0 ? 1 : caps[i - 1];
      const prevVal = i === 0 ? { mora: 0, wits: 0 } : table[prevCap];
      const curVal = table[cap];
      const ratio = (level - prevCap) / Math.max(1, cap - prevCap);
      return {
        mora: Math.round(prevVal.mora + (curVal.mora - prevVal.mora) * ratio),
        wits: Math.round(prevVal.wits + (curVal.wits - prevVal.wits) * ratio)
      };
    }
  }
  return table[caps[caps.length - 1]];
}

function addWeaponRequirements(addReq, weaponKey, curLvlRaw, tgtLvlRaw) {
  if (!weaponKey) return;
  const curLvl = Math.max(1, Math.min(90, Number(curLvlRaw) || 1));
  const tgtLvl = Math.max(curLvl, Math.min(90, Number(tgtLvlRaw) || 90));
  if (tgtLvl <= curLvl) return;

  const wm = getWeaponMeta(weaponKey);
  const rarity = Number(wm.rarity) || 5;
  const is5Star = rarity >= 5;

  // Total 1->90 EXP & ores: 5-star ~907 Mystic Ores / 906,400 Mora; 4-star ~605 Ores / 604,300 Mora
  const maxOres = is5Star ? 907 : rarity === 4 ? 605 : 403;
  const maxExpMora = is5Star ? 906400 : rarity === 4 ? 604300 : 402800;
  const ratio = (tgtLvl - curLvl) / 89;
  addReq('MysticEnhancementOre', Math.ceil(maxOres * ratio));
  addReq('Mora', Math.round(maxExpMora * ratio));

  const ascCaps = [20, 40, 50, 60, 70, 80];
  const ascMora = is5Star
    ? [10000, 20000, 30000, 45000, 55000, 65000]
    : [5000, 15000, 20000, 30000, 35000, 45000];
  const curPhase = ascCaps.filter((c) => curLvl > c).length;
  const tgtPhase = ascCaps.filter((c) => tgtLvl > c).length;
  for (let p = curPhase; p < tgtPhase; p++) {
    addReq('Mora', ascMora[p] || 0);
  }
  if (wm.domainMatExactKey && tgtPhase > curPhase) {
    const goldPerPhase = is5Star ? [0, 0, 0, 0, 2, 4] : [0, 0, 0, 0, 1, 3];
    let goldCount = 0;
    for (let p = curPhase; p < tgtPhase; p++) {
      goldCount += goldPerPhase[p] || 0;
    }
    if (goldCount > 0) addReq(wm.domainMatExactKey, goldCount);
  }
}

function computeRawTargetRequirements(target) {
  const catalog = getCatalog();
  const curves = catalog?.costCurves || {};
  const charMeta = getCharacterMeta(target.characterKey);

  const curLvl = Math.max(1, Math.min(95, Number(target.currentLevel) || 1));
  const tgtLvl = Math.max(curLvl, Math.min(95, Number(target.targetLevel) || 90));

  const req = {};
  const addReq = (matKey, amount) => {
    if (!matKey || !amount || amount <= 0) return;
    req[matKey] = (req[matKey] || 0) + amount;
  };

  // 1. Character EXP + EXP Mora
  const curExp = expWitsForLevel(curLvl, curves);
  const tgtExp = expWitsForLevel(tgtLvl, curves);
  addReq('HerosWit', Math.max(0, tgtExp.wits - curExp.wits));
  addReq('Mora', Math.max(0, tgtExp.mora - curExp.mora));

  // 2. Ascension Phases
  const curPhase = ascensionPhaseForLevel(curLvl);
  const tgtPhase = ascensionPhaseForLevel(tgtLvl);
  const ascCurve = curves.ascensionByPhase || {
    1: { mora: 20000, boss: 0, specialty: 3 },
    2: { mora: 40000, boss: 2, specialty: 10 },
    3: { mora: 60000, boss: 4, specialty: 20 },
    4: { mora: 80000, boss: 8, specialty: 30 },
    5: { mora: 100000, boss: 12, specialty: 45 },
    6: { mora: 120000, boss: 20, specialty: 60 }
  };

  for (let p = curPhase + 1; p <= tgtPhase; p++) {
    const step = ascCurve[p];
    if (!step) continue;
    addReq('Mora', step.mora || 0);
    if (charMeta.bossMat && step.boss) addReq(charMeta.bossMat, step.boss);
    if (charMeta.localSpecialty && step.specialty) addReq(charMeta.localSpecialty, step.specialty);
  }

  // 3. Talents (Auto / Skill / Burst)
  const talentCurve = curves.talentByLevel || {
    2: { mora: 12500, bookTier: 'Teachings', bookCount: 3, weekly: 0, crown: 0 },
    3: { mora: 17500, bookTier: 'Guide', bookCount: 2, weekly: 0, crown: 0 },
    4: { mora: 25000, bookTier: 'Guide', bookCount: 4, weekly: 0, crown: 0 },
    5: { mora: 30000, bookTier: 'Guide', bookCount: 6, weekly: 0, crown: 0 },
    6: { mora: 37500, bookTier: 'Guide', bookCount: 9, weekly: 0, crown: 0 },
    7: { mora: 120000, bookTier: 'Philosophies', bookCount: 4, weekly: 1, crown: 0 },
    8: { mora: 260000, bookTier: 'Philosophies', bookCount: 6, weekly: 1, crown: 0 },
    9: { mora: 450000, bookTier: 'Philosophies', bookCount: 12, weekly: 2, crown: 0 },
    10: { mora: 700000, bookTier: 'Philosophies', bookCount: 16, weekly: 2, crown: 1 }
  };

  const bookBase = charMeta.talentBook || 'Gold';
  const slots = ['auto', 'skill', 'burst'];
  for (const slot of slots) {
    const cT = Math.max(1, Math.min(10, Number(target.currentTalent?.[slot]) || 1));
    const tT = Math.max(cT, Math.min(10, Number(target.targetTalent?.[slot]) || cT));
    for (let lvl = cT + 1; lvl <= tT; lvl++) {
      const step = talentCurve[lvl];
      if (!step) continue;
      addReq('Mora', step.mora || 0);
      const prefix = step.bookTier === 'Teachings' ? 'TeachingsOf' : step.bookTier === 'Guide' ? 'GuideTo' : 'PhilosophiesOf';
      addReq(`${prefix}${bookBase}`, step.bookCount || 0);
      if (step.weekly && charMeta.weeklyMat) addReq(charMeta.weeklyMat, step.weekly);
      if (step.crown) addReq('CrownOfInsight', step.crown);
    }
  }

  // 4. Optional paired Weapon target
  if (target.weaponKey) {
    addWeaponRequirements(addReq, target.weaponKey, target.weaponCurrentLevel, target.weaponTargetLevel);
  }

  return req;
}

function formatRequirementRows(req, getOwnedForMat) {
  const rows = Object.entries(req).map(([matKey, needed]) => {
    const owned = getOwnedForMat(matKey);
    const delta = owned - needed;
    const meta = getMaterialMeta(matKey);
    return {
      key: matKey,
      name: meta.name,
      category: meta.category,
      needed,
      owned,
      delta, // negative = deficit, >= 0 = surplus
      deficit: delta < 0 ? Math.abs(delta) : 0,
      surplus: delta >= 0 ? delta : 0
    };
  });

  // Sort deficits first, then by needed descending
  rows.sort((a, b) => {
    if (a.deficit > 0 && b.deficit === 0) return -1;
    if (a.deficit === 0 && b.deficit > 0) return 1;
    return b.deficit - a.deficit || b.needed - a.needed;
  });

  return rows;
}

/**
 * Calculates exact material requirements & deficits for a single prefarm target
 * against catalog.json recipes and current inventory.
 */
export function calculatePrefarmRequirements(target, inventory = {}, wallet = null) {
  const req = computeRawTargetRequirements(target);
  const effectiveWits = Math.floor(calcEffectiveHerosWit(inventory).effectiveTotal);
  return formatRequirementRows(req, (matKey) =>
    matKey === 'HerosWit'
      ? effectiveWits
      : matKey === 'Mora'
        ? Number(wallet?.mora ?? inventory[matKey]?.count ?? 0)
        : Number(inventory[matKey]?.count ?? 0)
  );
}

/**
 * Sorts prefarmQueue by (tier, rank) and assigns sequential global rank plus
 * per-tier subRank (e.g. #1.1, #1.2, #2.1). Legacy items without `tier`
 * default to `tier = rank`.
 */
export function sortAndLabelPrefarmQueue(prefarmQueue = []) {
  const sorted = [...prefarmQueue].sort((a, b) => {
    const aTier = Math.max(1, Number(a.tier) || Number(a.rank) || 1);
    const bTier = Math.max(1, Number(b.tier) || Number(b.rank) || 1);
    if (aTier !== bTier) return aTier - bTier;
    return (Number(a.rank) || 0) - (Number(b.rank) || 0);
  });
  const tierCounts = new Map();
  return sorted.map((t, idx) => {
    const tier = Math.max(1, Number(t.tier) || Number(t.rank) || 1);
    const subRank = (tierCounts.get(tier) || 0) + 1;
    tierCounts.set(tier, subRank);
    return {
      ...t,
      tier,
      subRank,
      rank: idx + 1,
      tierLabel: `#${tier}.${subRank}`
    };
  });
}

/**
 * Waterfall allocation across the ranked prefarmQueue (#1.1 -> #1.2 -> #2.1 ...).
 * Higher-priority targets consume shared resources (Mora, Hero's Wit, Crowns, Books, Weapon mats)
 * first, and only the remaining surplus flows down to lower-priority targets.
 */
export function calculateWaterfallPrefarm(prefarmQueue = [], inventory = {}, wallet = null) {
  const sorted = sortAndLabelPrefarmQueue(prefarmQueue);
  const effectiveWits = Math.floor(calcEffectiveHerosWit(inventory).effectiveTotal);
  const pool = new Map();

  const getPool = (matKey) => {
    if (pool.has(matKey)) return pool.get(matKey);
    const initial =
      matKey === 'HerosWit'
        ? effectiveWits
        : matKey === 'Mora'
          ? Number(wallet?.mora ?? inventory[matKey]?.count ?? 0)
          : Number(inventory[matKey]?.count ?? 0);
    pool.set(matKey, initial);
    return initial;
  };

  const results = new Map();
  for (const target of sorted) {
    const req = computeRawTargetRequirements(target);
    const rows = formatRequirementRows(req, (matKey) => getPool(matKey));
    for (const [matKey, needed] of Object.entries(req)) {
      pool.set(matKey, Math.max(0, getPool(matKey) - needed));
    }
    results.set(target.id, rows);
  }
  return results;
}

const WEEKDAY_SHORT = {
  Monday: 'Mon',
  Tuesday: 'Tue',
  Wednesday: 'Wed',
  Thursday: 'Thu',
  Friday: 'Fri',
  Saturday: 'Sat',
  Sunday: 'Sun'
};

function getDomainInfoForReward(matKey, todayName) {
  const catalog = getCatalog();
  const normTarget = normalizeAssetKeySimple(matKey);
  const schedule = catalog?.domainSchedule || {};
  const days = [];
  let domainName = '';
  let seriesShortName = '';
  let domainType = '';

  for (const [dayFull, list] of Object.entries(schedule)) {
    for (const dom of list || []) {
      const match = (dom.rewards || []).find((r) => normalizeAssetKeySimple(r.key) === normTarget);
      if (match) {
        if (!domainName) domainName = dom.name;
        if (!domainType) domainType = dom.domainType;
        if (!seriesShortName) seriesShortName = match.name;
        const shortDay = WEEKDAY_SHORT[dayFull] || dayFull.slice(0, 3);
        if (!days.includes(shortDay)) days.push(shortDay);
      }
    }
  }

  const todayShort = todayName ? WEEKDAY_SHORT[todayName] || todayName.slice(0, 3) : '';
  return {
    domainName,
    domainType,
    seriesShortName,
    days,
    isToday: Boolean(todayShort && days.includes(todayShort))
  };
}

function normalizeAssetKeySimple(str) {
  return String(str || '').toLowerCase().replace(/[^a-z0-9]/g, '');
}

/**
 * Aggregates & transforms the shared resource pool for Priority #1 & #2 targets
 * into an actionable farming and resin planner (grouped tiered series with 3:1
 * alchemy crafting, EXP pool, WL9 resin/run estimates, and #1/#2 allocation).
 */
export function calculatePrioritySharedPool(prefarmQueue = [], inventory = {}, wallet = null, todayName = null) {
  const labeled = sortAndLabelPrefarmQueue(prefarmQueue);
  const activeTiers = [...new Set(labeled.map((t) => t.tier))].slice(0, 2);
  const sorted = labeled.filter((t) => activeTiers.includes(t.tier));
  if (!sorted.length) return [];

  const effectiveWits = Math.floor(calcEffectiveHerosWit(inventory).effectiveTotal);
  const expBreakdown = calcEffectiveHerosWit(inventory);
  const dreamSolventCount = Number(inventory.DreamSolvent?.count ?? 0);

  const getRawOwned = (matKey) => {
    if (matKey === 'HerosWit') return effectiveWits;
    if (matKey === 'Mora') return Number(wallet?.mora ?? inventory[matKey]?.count ?? 0);
    return Number(inventory[matKey]?.count ?? 0);
  };

  // Track per-target raw requirements and classify each material key
  const targetReqs = sorted.map((t) => ({
    target: t,
    charMeta: getCharacterMeta(t.characterKey),
    weaponMeta: t.weaponKey ? getWeaponMeta(t.weaponKey) : null,
    req: computeRawTargetRequirements(t)
  }));

  const matKindByKey = new Map();
  const talentSeriesMap = new Map(); // base -> { base, byTier: { T2, T3, T4 }, byTarget: Map<rank, {T2,T3,T4}> }
  const flatMap = new Map(); // matKey -> { key, kind, neededByRank: Map<rank, number> }

  for (const { target, charMeta, weaponMeta, req } of targetReqs) {
    for (const [matKey, needed] of Object.entries(req)) {
      if (!needed || needed <= 0) continue;

      const bookMatch = matKey.match(/^(TeachingsOf|GuideTo|PhilosophiesOf)(.+)$/);
      if (bookMatch) {
        const tierPrefix = bookMatch[1];
        const base = bookMatch[2];
        const tierKey = tierPrefix === 'TeachingsOf' ? 'T2' : tierPrefix === 'GuideTo' ? 'T3' : 'T4';
        if (!talentSeriesMap.has(base)) {
          talentSeriesMap.set(base, {
            base,
            keys: {
              T2: `TeachingsOf${base}`,
              T3: `GuideTo${base}`,
              T4: `PhilosophiesOf${base}`
            },
            neededByTier: { T2: 0, T3: 0, T4: 0 },
            byRank: new Map()
          });
        }
        const series = talentSeriesMap.get(base);
        series.neededByTier[tierKey] += needed;
        if (!series.byRank.has(target.rank)) series.byRank.set(target.rank, { T2: 0, T3: 0, T4: 0 });
        series.byRank.get(target.rank)[tierKey] += needed;
        continue;
      }

      // Determine kind for non-book materials
      let kind = matKindByKey.get(matKey);
      if (!kind) {
        if (matKey === 'HerosWit') kind = 'exp';
        else if (matKey === 'Mora') kind = 'mora';
        else if (matKey === 'CrownOfInsight') kind = 'crown';
        else if (matKey === 'MysticEnhancementOre') kind = 'ore';
        else if (weaponMeta?.domainMatExactKey && matKey === weaponMeta.domainMatExactKey) kind = 'weaponSeries';
        else if (charMeta?.weeklyMat && matKey === charMeta.weeklyMat) kind = 'weeklyBoss';
        else if (charMeta?.bossMat && matKey === charMeta.bossMat) kind = 'worldBoss';
        else if (charMeta?.localSpecialty && matKey === charMeta.localSpecialty) kind = 'localSpecialty';
        else kind = 'other';
        matKindByKey.set(matKey, kind);
      }

      if (!flatMap.has(matKey)) {
        flatMap.set(matKey, { key: matKey, kind, neededTotal: 0, byRank: new Map() });
      }
      const entry = flatMap.get(matKey);
      entry.neededTotal += needed;
      entry.byRank.set(target.rank, (entry.byRank.get(target.rank) || 0) + needed);
    }
  }

  const buildAlloc = (needByRankEquiv, totalNeedEquiv, netDeficitEquiv) => {
    const fundedTotal = Math.max(0, totalNeedEquiv - netDeficitEquiv);
    let remaining = fundedTotal;
    const targets = [];
    targetReqs.forEach(({ target, charMeta }, idx) => {
      const charNeed = needByRankEquiv.get(target.rank) || 0;
      if (charNeed <= 0) return;
      const funded = Math.min(charNeed, remaining);
      remaining = Math.max(0, remaining - funded);
      targets.push({
        rank: targets.length + 1,
        slotIndex: idx,
        tier: target.tier,
        tierLabel: target.tierLabel,
        characterKey: target.characterKey,
        name: charMeta.name || target.characterKey,
        labelPrefix: `${target.tierLabel} `,
        needed: charNeed,
        funded,
        pct: Math.round((funded / charNeed) * 100),
        widthPct: totalNeedEquiv > 0 ? Number(((funded / totalNeedEquiv) * 100).toFixed(2)) : 0
      });
    });
    return { targets };
  };

  const results = [];

  // 1. Talent Book Series (with 3-to-1 Alchemy Conversion)
  for (const series of talentSeriesMap.values()) {
    const o2 = getRawOwned(series.keys.T2);
    const o3 = getRawOwned(series.keys.T3);
    const o4 = getRawOwned(series.keys.T4);
    const n2 = series.neededByTier.T2;
    const n3 = series.neededByTier.T3;
    const n4 = series.neededByTier.T4;

    const surplus2 = Math.max(0, o2 - n2);
    const craft3 = Math.floor(surplus2 / 3);
    const netDef2 = Math.max(0, n2 - o2);

    const eff3 = o3 + craft3;
    const surplus3 = Math.max(0, eff3 - n3);
    const craft4 = Math.floor(surplus3 / 3);
    const netDef3 = Math.max(0, n3 - eff3);

    const eff4 = o4 + craft4;
    const surplus4 = Math.max(0, eff4 - n4);
    const netDef4 = Math.max(0, n4 - eff4);

    const deficit = netDef2 + netDef3 + netDef4;
    const goldEquivDeficit = netDef4 + netDef3 / 3 + netDef2 / 9;
    const surplus = deficit === 0 ? surplus4 : 0;

    const domInfo = getDomainInfoForReward(series.keys.T4, todayName);
    const needEquivByRank = new Map();
    let totalNeedEquiv = 0;
    for (const [rank, t] of series.byRank.entries()) {
      const eq = t.T4 + t.T3 / 3 + t.T2 / 9;
      needEquivByRank.set(rank, eq);
      totalNeedEquiv += eq;
    }

    let farmingCost = null;
    if (deficit > 0) {
      const runs = Math.max(1, Math.ceil(goldEquivDeficit / 2.2));
      const resin = runs * 20;
      farmingCost = { runs, resin, text: `~${runs} Runs · ${resin.toLocaleString()} Resin` };
    }

    results.push({
      key: series.keys.T4,
      name: `${series.base} Series`,
      kind: 'talentSeries',
      rarity: 4,
      owned: eff4,
      neededTotal: n4 + n3 + n2,
      deficit,
      surplus,
      tiers: [
        { label: 'T2', key: series.keys.T2, owned: o2, needed: n2, netDeficit: netDef2 },
        { label: 'T3', key: series.keys.T3, owned: eff3, rawOwned: o3, craftedIn: craft3, needed: n3, netDeficit: netDef3 },
        { label: 'T4', key: series.keys.T4, owned: eff4, rawOwned: o4, craftedIn: craft4, needed: n4, netDeficit: netDef4 }
      ],
      sourceLabel: domInfo.domainName || 'Talent Domain',
      scheduleDays: domInfo.days,
      isToday: domInfo.isToday,
      farmingCost,
      urgency: domInfo.isToday ? 2 : 3,
      alloc: buildAlloc(needEquivByRank, totalNeedEquiv, goldEquivDeficit)
    });
  }

  // 2. Non-Book Materials (Weapon Mats, Weekly Boss, World Boss, EXP, Mora, Local Specialty, etc.)
  for (const item of flatMap.values()) {
    const meta = getMaterialMeta(item.key);
    const owned = getRawOwned(item.key);
    const delta = owned - item.neededTotal;
    const deficit = delta < 0 ? Math.abs(delta) : 0;
    const surplus = delta >= 0 ? delta : 0;

    let name = meta.name;
    let sourceLabel = 'Material';
    let scheduleDays = [];
    let isToday = false;
    let urgency = 4;
    let farmingCost = null;
    let tiers = null;

    if (item.kind === 'weaponSeries') {
      const domInfo = getDomainInfoForReward(item.key, todayName);
      name = domInfo.seriesShortName ? `${domInfo.seriesShortName} Series` : meta.name;
      sourceLabel = domInfo.domainName || 'Weapon Domain';
      scheduleDays = domInfo.days;
      isToday = domInfo.isToday;
      urgency = domInfo.isToday ? 2 : 3;
      tiers = [{ label: 'T5', key: item.key, owned, needed: item.neededTotal, netDeficit: deficit }];
      if (deficit > 0) {
        const runs = Math.max(1, Math.ceil(deficit / 2.2));
        const resin = runs * 20;
        farmingCost = { runs, resin, text: `~${runs} Runs · ${resin.toLocaleString()} Resin` };
      }
    } else if (item.kind === 'weeklyBoss') {
      sourceLabel = 'Weekly Boss';
      urgency = 1;
      if (deficit > 0) {
        const weeks = Math.max(1, Math.ceil(deficit / 2.4));
        farmingCost = { weeks, text: `~${weeks} ${weeks === 1 ? 'Week' : 'Weeks'}` };
      }
    } else if (item.kind === 'worldBoss') {
      sourceLabel = meta.bossName ? `Boss · ${meta.bossName}` : 'Overworld Boss';
      urgency = 4;
      if (deficit > 0) {
        const runs = Math.max(1, Math.ceil(deficit / 2.5));
        const resin = runs * 40;
        farmingCost = { runs, resin, text: `~${runs} Runs · ${resin.toLocaleString()} Resin` };
      }
    } else if (item.kind === 'exp') {
      name = "Hero's Wit";
      sourceLabel = 'Ley Line';
      urgency = 4;
      if (deficit > 0) {
        const runs = Math.max(1, Math.ceil((deficit * 20000) / 122500));
        const resin = runs * 20;
        farmingCost = { runs, resin, text: `~${runs} Runs · ${resin.toLocaleString()} Resin` };
      }
    } else if (item.kind === 'mora') {
      sourceLabel = 'Ley Line';
      urgency = 4;
      if (deficit > 0) {
        const runs = Math.max(1, Math.ceil(deficit / 120000));
        const resin = runs * 20;
        farmingCost = { runs, resin, text: `~${runs} Runs · ${resin.toLocaleString()} Resin` };
      }
    } else if (item.kind === 'localSpecialty') {
      sourceLabel = 'Local Specialty';
    } else if (item.kind === 'crown') {
      sourceLabel = 'Event / Limited';
    } else if (item.kind === 'ore') {
      sourceLabel = 'Forging / Mining';
    }

    results.push({
      key: item.key,
      name,
      kind: item.kind,
      rarity: meta.rarity || (item.kind === 'crown' ? 5 : item.kind === 'weeklyBoss' ? 5 : item.kind === 'exp' || item.kind === 'worldBoss' ? 4 : 3),
      owned,
      neededTotal: item.neededTotal,
      deficit,
      surplus,
      tiers,
      expBreakdown: item.kind === 'exp' ? expBreakdown : null,
      sourceLabel,
      scheduleDays,
      isToday,
      bossUnreleased: Boolean(meta.bossUnreleased),
      dreamSolventCount: item.kind === 'weeklyBoss' ? dreamSolventCount : 0,
      farmingCost,
      urgency,
      alloc: buildAlloc(item.byRank, item.neededTotal, deficit)
    });
  }

  return results.sort((a, b) => {
    const aDef = a.deficit > 0 ? 0 : 1;
    const bDef = b.deficit > 0 ? 0 : 1;
    if (aDef !== bDef) return aDef - bDef;
    if (a.urgency !== b.urgency) return a.urgency - b.urgency;
    return b.deficit - a.deficit || b.neededTotal - a.neededTotal;
  });
}

/**
 * Detects weapon instance conflicts across all Team Comps.
 */
export function detectTeamWeaponConflicts(teams = []) {
  const usageMap = new Map(); // weaponId -> Array<{ teamId, teamName, characterKey, weaponKey }>

  for (const team of teams) {
    for (const member of team.members || []) {
      const instanceKey = member.weaponId || member.weaponKey;
      if (!instanceKey) continue;
      if (!usageMap.has(instanceKey)) usageMap.set(instanceKey, []);
      usageMap.get(instanceKey).push({
        teamId: team.id,
        teamName: team.name,
        characterKey: member.characterKey,
        weaponKey: member.weaponKey
      });
    }
  }

  const conflicts = [];
  for (const [instanceKey, usages] of usageMap.entries()) {
    const uniqueTeams = new Set(usages.map((u) => u.teamId));
    if (uniqueTeams.size > 1) {
      conflicts.push({
        instanceKey,
        weaponKey: usages[0].weaponKey,
        usages
      });
    }
  }
  return conflicts;
}

              }
            }
          }
        }
        slim._lunarisVer = lunarisVer;
        slim._matIdLookup = matIdLookup;

            slim.characters[norm] = {

        if (lWepRes.status === 'fulfilled' && lWepRes.value) {
          for (const [wepId, item] of Object.entries(lWepRes.value)) {

            slim.weapons[norm] = {

              pascalKey: String(name).replace(/[^a-zA-Z0-9]/g, ''),

              rarity: item.qualityType === 'QUALITY_ORANGE' ? 5 : item.qualityType === 'QUALITY_PURPLE' ? 4 : 3,

      } catch {
        // Ignore Lunaris fallback errors
      }
    }

    if (!Object.keys(slim.characters).length && !Object.keys(slim.weapons).length) {

    }

    catalogSource = 'live';
    catalogSyncedAt = new Date().toISOString();
    lastCacheTs = Date.now();
    try {

        }
      }
      localStorage.setItem(YATTA_CACHE_KEY, JSON.stringify({ ts: lastCacheTs, data: slim }));
    } catch {
      // Ignore quota errors
    }
  } catch (err) {
    console.warn('Project Amber (gi.yatta.moe) asset fetch failed, trying local snapshot:', err);
    await loadSnapshotFallback();
  }
}

  if (pendingDetailFetches.has(norm)) return pendingDetailFetches.get(norm);

  const task = (async () => {
    try {
      const url = `https://api.lunaris.moe/data/${encodeURIComponent(yattaMaps.lunarisVer)}/en/char/${encodeURIComponent(entry.lunarisId)}.json`;
      const res = await fetch(url);
      if (!res.ok) {
        entry._hydrated = true;
        return false;
      }
      const detail = await res.json();
      entry._hydrated = true;
      const lookup = yattaMaps.matIdLookup || {};
      const toPascal = (s) => String(s || '').replace(/[^a-zA-Z0-9]/g, '');

      const booksObj = detail?.skills?.leveling?.books || {};
      for (const matId of Object.keys(booksObj)) {
        const matName = lookup[String(matId)];
        if (matName && /^Philosophies of /i.test(matName)) {
          entry.talentBook = matName.replace(/^Philosophies of\s+/i, '');
          entry.domainMatKey = normalizeAssetKey(matName);
          break;
        }
      }

      const lastAsc = Array.isArray(detail?.ascension)
        ? detail.ascension[detail.ascension.length - 1]
        : detail?.ascension || null;
      const bossId = lastAsc?.elitemonster ? Object.keys(lastAsc.elitemonster)[0] : null;
      if (bossId && lookup[bossId]) entry.bossMat = toPascal(lookup[bossId]);

      const specId = lastAsc?.speciality ? Object.keys(lastAsc.speciality)[0] : null;
      if (specId && lookup[specId]) entry.localSpecialty = toPascal(lookup[specId]);

      const weeklyObj = detail?.skills?.leveling?.weekly || {};
      const weeklyId = Object.keys(weeklyObj)[0];
      if (weeklyId && lookup[weeklyId]) entry.weeklyMat = toPascal(lookup[weeklyId]);

      // Persist updated entry back to localStorage cache
      try {
        const cached = JSON.parse(localStorage.getItem(YATTA_CACHE_KEY) || '{}');
        if (cached?.data?.characters?.[norm]) {
          cached.data.characters[norm] = entry;
          localStorage.setItem(YATTA_CACHE_KEY, JSON.stringify(cached));
        }
      } catch {
        // Ignore storage errors
      }
      return Boolean(entry.talentBook || entry.bossMat || entry.localSpecialty);
    } catch {
      return false;
    } finally {
      pendingDetailFetches.delete(norm);
    }
  })();

  pendingDetailFetches.set(norm, task);
  return task;
}

  // Materials
  const nextInventory = { ...(account.inventory || {}) };
  const matChanges = [];
  for (const [matKey, rawCount] of Object.entries(good.materials || {})) {
    const count = Number(rawCount) || 0;
    const prevEntry = nextInventory[matKey];
    const prevCount = prevEntry?.count ?? null;
    if (prevCount !== count) {
      matChanges.push({
        key: matKey,
        from: prevCount,
        to: count
      });
    }
    const meta = getMaterialMeta(matKey);
    nextInventory[matKey] = {
      count,
      target: prevEntry?.target ?? 0,
      category: prevEntry?.category || meta.category || 'Material'
    };
  }

  const nextWallet = { ...(account.wallet || {}) };
  if (typeof good.materials?.Mora === 'number') {
    nextWallet.mora = good.materials.Mora;
  }
  if (typeof good.materials?.IntertwinedFate === 'number') {
    nextWallet.intertwinedFates = good.materials.IntertwinedFate;
  }
  if (typeof good.materials?.AcquaintFate === 'number') {
    nextWallet.acquaintFates = good.materials.AcquaintFate;
  }

  const mergedAccount = {
    ...account,
    wallet: nextWallet,
    characters: Array.from(mergedCharsMap.values()),
    weapons: nextWeapons.length ? nextWeapons : account.weapons,
    inventory: nextInventory
  };

  return {
    source: good.source || 'GOOD v3',
    version: good.version,
    charactersTotal: (good.characters || []).length,
    weaponsTotal: (good.weapons || []).length,
    materialsTotal: Object.keys(good.materials || {}).length,
    charChanges,
    weaponChanges,
    matChanges,
    mergedAccount
  };
}

export async function commitGoodImport(username, goodPayload, mergedAccount) {
  return postImportGoodJson(username, goodPayload, mergedAccount);
}

function formatCompactSurplus(n) {
  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(2).replace(/\.?0+$/, '')}M`;
  if (n >= 10_000) return `${(n / 1_000).toFixed(1).replace(/\.0$/, '')}K`;
  return n.toLocaleString();
}

function formatCompactNum(n) {
  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(2).replace(/\.?0+$/, '')}M`;
  if (n >= 100_000) return `${(n / 1_000).toFixed(0)}K`;
  return Math.round(n).toLocaleString();
}

function escapeHtml(str) {
  return String(str ?? '')
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;');
}

function isSwapFlagged(c) {
  return Boolean((c.swapTag && c.swapTag.trim()) || (c.notes && SWAP_NOTE_RE.test(c.notes)));
}

function formatStatusLabel(c) {
  if (c.status === 'Active Prefarm') {
    return c.priorityRank ? `#${c.priorityRank} PREFARM` : 'PREFARM';
  }
  if (c.status === 'In Progress') return 'BUILDING';
  if (c.status === 'Built') return 'BUILT';
  if (c.status === 'Bench') return 'BENCH';
  return 'UNOWNED';
}

function getStatusPillClass(status) {
  if (status === 'Built') return 'pill-mint';
  if (status === 'Active Prefarm') return 'pill-gold';
  if (status === 'In Progress') return 'pill-sky';
  if (status === 'Unowned') return 'pill-muted';
  return '';
}

function sortByPriorityThenLevel(a, b) {
  const ra = Number(a.priorityRank) || 9999;
  const rb = Number(b.priorityRank) || 9999;
  if (ra !== rb) return ra - rb;
  return b.level - a.level || a.meta.name.localeCompare(b.meta.name);
}

function renderCharacterCard(c, density) {
  const cornerBadge = c.owned ? `Lv.${c.level} · C${c.constellation}` : 'UNOWNED';
  const statusLabel = formatStatusLabel(c);
  const statusClass = getStatusPillClass(c.status);
  const swapBadge = isSwapFlagged(c) ? '<span class="roster-swap-flag mono" title="Gear Swap / Borrowing Flagged">⇄</span>' : '';

  if (density === 'list') {
    return `
      <button type="button" class="roster-list-row ${!c.owned ? 'is-unowned' : ''}" data-inspect-char="${c.key}">
        <span class="roster-list-left">
          ${renderAssetIcon('characters', c.key, c.meta.name, 'icon-sm')}
          <span class="roster-list-name">${escapeHtml(c.meta.name)}</span>
        </span>
        <span class="roster-list-right mono">
          ${swapBadge}
          <span class="roster-corner-badge">${cornerBadge}</span>
          <span class="pill ${statusClass}">${statusLabel}</span>
        </span>
      </button>`;
  }

  return `
    <button type="button" class="roster-portrait-card ${!c.owned ? 'is-unowned' : ''}" data-inspect-char="${c.key}">
      <span class="roster-corner-badge mono">${cornerBadge}</span>
      <div class="roster-portrait-frame">
        ${renderAssetIcon('characters', c.key, c.meta.name, 'icon-portrait')}
      </div>
      <div class="roster-portrait-footer">
        <span class="roster-portrait-name">${escapeHtml(c.meta.name)}</span>
        <span class="roster-portrait-pills">
          <span class="pill mono ${statusClass}">${statusLabel}</span>
          ${swapBadge}
        </span>
      </div>
    </button>`;
}

function renderTierSection(tierId, title, list, isOpen, density) {
  const body = list.length
    ? `<div class="${density === 'list' ? 'roster-compact-list' : 'roster-portrait-grid'}">
        ${list.map((c) => renderCharacterCard(c, density)).join('')}
       </div>`
    : '<div class="empty-state">No characters in this tier match the current filters.</div>';

  return `
    <details class="roster-tier" data-tier-id="${tierId}" ${isOpen ? 'open' : ''}>
      <summary class="roster-tier-header">
        <span class="roster-tier-title">${title}</span>
        <span class="pill mono">${list.length}</span>
      </summary>
      <div class="roster-tier-body">${body}</div>
    </details>`;
}

export function renderRosterTab(container, state, handlers) {
  const { account, isReadOnly, rosterFilter } = state;
  const ownedMap = new Map((account.characters || []).map((c) => [c.key, c]));
  const prefarmSet = new Set((account.prefarmQueue || []).map((p) => p.characterKey));

  const weaponByLocation = new Map();
  for (const w of account.weapons || []) {
    if (w.location) weaponByLocation.set(w.location, w);
  }

  const allKeys = new Set([
    ...ownedMap.keys(),
    ...getAllCharacterKeys()
  ]);

  const combinedChars = Array.from(allKeys).map((key) => {
    const meta = getCharacterMeta(key);
    const owned = ownedMap.get(key);
    if (!owned || owned.status === 'Unowned' || owned.owned === false) {
      return {
        key,
        meta,
        owned: false,
        level: owned?.level || 0,
        targetLevel: owned?.targetLevel || 90,
        constellation: owned?.constellation || 0,
        talent: owned?.talent || { auto: 1, skill: 1, burst: 1 },
        status: 'Unowned',
        priorityRank: owned?.priorityRank || '',
        swapTag: owned?.swapTag || '',
        notes: owned?.notes || '',
        equippedWeapon: null
      };
    }

    const effectiveStatus =
      prefarmSet.has(key) && owned.status !== 'Built'
        ? 'Active Prefarm'
        : owned.status || 'In Progress';

    return {
      key,
      meta,
      owned: true,
      level: Number(owned.level) || 1,
      targetLevel: Number(owned.targetLevel) || 90,
      constellation: Number(owned.constellation) || 0,
      talent: owned.talent || { auto: 1, skill: 1, burst: 1 },
      status: effectiveStatus,
      priorityRank: owned.priorityRank || '',
      swapTag: owned.swapTag || '',
      notes: owned.notes || '',
      equippedWeapon: weaponByLocation.get(key) || null
    };
  });

  const q = (rosterFilter.search || '').trim().toLowerCase();
  const statusFilter = rosterFilter.status || 'All';
  const elementFilter = rosterFilter.element || 'All';
  const density = rosterFilter.density === 'list' ? 'list' : 'grid';
  const tiersState = rosterFilter.tiers || { active: true, built: true, archive: false };

  const filteredChars = combinedChars.filter((c) => {
    if (statusFilter === 'Active Work' && c.status !== 'Active Prefarm' && c.status !== 'In Progress') return false;
    if (statusFilter === 'Combat Ready' && c.status !== 'Built') return false;
    if (statusFilter === 'Swap Flagged' && !isSwapFlagged(c)) return false;
    if (elementFilter !== 'All' && c.meta.element !== elementFilter) return false;
    if (q && !c.meta.name.toLowerCase().includes(q) && !c.key.toLowerCase().includes(q)) return false;
    return true;
  });

  const activeTier = filteredChars
    .filter((c) => c.status === 'Active Prefarm' || c.status === 'In Progress')
    .sort(sortByPriorityThenLevel);

  const builtTier = filteredChars
    .filter((c) => c.status === 'Built')
    .sort((a, b) => b.level - a.level || a.meta.name.localeCompare(b.meta.name));

  const archiveTier = filteredChars
    .filter((c) => c.status === 'Bench' || c.status === 'Unowned')
    .sort((a, b) => (a.owned === b.owned ? b.level - a.level || a.meta.name.localeCompare(b.meta.name) : a.owned ? -1 : 1));

  const filteredWeapons = [...(account.weapons || [])]
    .filter((w) => {
      if (!q) return true;
      const wm = getWeaponMeta(w.key);
      return wm.name.toLowerCase().includes(q) || w.key.toLowerCase().includes(q) || (w.location || '').toLowerCase().includes(q);
    })
    .sort((a, b) => b.level - a.level || b.refinement - a.refinement);

  const inspectedChar = rosterFilter.inspectedKey
    ? combinedChars.find((c) => c.key === rosterFilter.inspectedKey) || null
    : null;

  container.innerHTML = `
    <div class="section-header">
      <div>
        <div class="eyebrow">AARUDB / ROSTER</div>
        <h1 class="section-title">Roster &amp; Armory</h1>
      </div>
      <div class="header-meta mono">
        OWNED: <span class="text-gold">${combinedChars.filter((c) => c.owned).length}</span> CHARS • <span class="text-mint">${(account.weapons || []).length}</span> WEAPONS
      </div>
    </div>

    <article class="card">
      <div class="roster-toolbar">
        <div class="roster-pill-row">
          <div class="filter-pills">
            ${STATUS_PILLS.map(
              (st) => `<button type="button" class="filter-pill ${statusFilter === st ? 'active' : ''}" data-roster-status="${st}">${st}</button>`
            ).join('')}
          </div>
          <span class="roster-pill-divider"></span>
          <div class="filter-pills">
            ${ELEMENT_PILLS.map(
              (el) => `<button type="button" class="filter-pill ${elementFilter === el ? 'active' : ''}" data-roster-element="${el}">${el === 'All' ? 'All Elements' : el}</button>`
            ).join('')}
          </div>
        </div>

        <div class="toolbar-right">
          <div class="subtab-switch">
            <button type="button" class="filter-pill ${density === 'grid' ? 'active' : ''}" data-roster-density="grid">Portrait Grid</button>
            <button type="button" class="filter-pill ${density === 'list' ? 'active' : ''}" data-roster-density="list">Compact List</button>
          </div>
          <div class="subtab-switch">
            <button type="button" class="filter-pill ${rosterFilter.view === 'characters' ? 'active' : ''}" data-roster-view="characters">Characters (${filteredChars.length})</button>
            <button type="button" class="filter-pill ${rosterFilter.view === 'weapons' ? 'active' : ''}" data-roster-view="weapons">Weapons (${filteredWeapons.length})</button>
          </div>
          <input type="search" class="input-search" placeholder="Search character or weapon..." value="${escapeHtml(rosterFilter.search || '')}" data-roster-search />
        </div>
      </div>

      ${
        rosterFilter.view === 'characters'
          ? `
        <div class="roster-tiers">
          ${renderTierSection('active', 'Active Prefarm & In Progress', activeTier, tiersState.active, density)}
          ${renderTierSection('built', 'Combat Ready / Built', builtTier, tiersState.built, density)}
          ${renderTierSection('archive', 'Bench & Unowned Archive', archiveTier, tiersState.archive, density)}
        </div>`
          : `
        <div class="roster-grid">
          ${filteredWeapons.length
            ? filteredWeapons
                .map((w) => {
                  const wm = getWeaponMeta(w.key);
                  const holderMeta = w.location ? getCharacterMeta(w.location) : null;
                  return `
                  <div class="roster-card">
                    <div class="roster-card-top">
                      <div class="roster-identity">
                        ${renderAssetIcon('weapons', w.key, wm.name, 'icon-md')}
                        <div>
                          <div class="item-name">${escapeHtml(wm.name)}</div>
                          <div class="item-meta mono">${escapeHtml(wm.weaponType)} • ${wm.rarity}★</div>
                        </div>
                      </div>
                      <span class="pill mono ${w.level >= 90 ? 'pill-mint' : ''}">Lv.${w.level}</span>
                    </div>
                    <div class="roster-stats mono">
                      <div class="stat-box">
                        <span class="stat-label">REFINEMENT</span>
                        <span class="stat-val text-gold">R${w.refinement}</span>
                      </div>
                      <div class="stat-box">
                        <span class="stat-label">ASC</span>
                        <span class="stat-val">A${w.ascension}</span>
                      </div>
                      <div class="stat-box">
                        <span class="stat-label">EQUIPPED</span>
                        <span class="stat-val">${holderMeta ? escapeHtml(holderMeta.name) : 'Unassigned'}</span>
                      </div>
                    </div>
                  </div>`;
                })
                .join('')
            : '<div class="empty-state">No weapons match the search query.</div>'}
        </div>`
      }
    </article>

    ${renderInspectorDrawer(inspectedChar, isReadOnly)}
  `;

  container.querySelectorAll('[data-roster-status]').forEach((btn) => {
    btn.addEventListener('click', () => handlers.onChangeRosterFilter({ status: btn.dataset.rosterStatus }));
  });

  container.querySelectorAll('[data-roster-element]').forEach((btn) => {
    btn.addEventListener('click', () => handlers.onChangeRosterFilter({ element: btn.dataset.rosterElement }));
  });

  container.querySelectorAll('[data-roster-density]').forEach((btn) => {
    btn.addEventListener('click', () => handlers.onChangeRosterFilter({ density: btn.dataset.rosterDensity }));
  });

  container.querySelectorAll('[data-roster-view]').forEach((btn) => {
    btn.addEventListener('click', () => handlers.onChangeRosterFilter({ view: btn.dataset.rosterView }));
  });

  container.querySelector('[data-roster-search]')?.addEventListener('input', (e) => {
    handlers.onChangeRosterFilter({ search: e.target.value }, { preserveFocusSelector: '[data-roster-search]' });
  });

  container.querySelectorAll('.roster-tier').forEach((det) => {
    det.addEventListener('toggle', () => {
      const id = det.dataset.tierId;
      handlers.onChangeRosterFilter(
        { tiers: { ...tiersState, [id]: det.open } },
        { silent: true }
      );
    });
  });

  container.querySelectorAll('[data-inspect-char]').forEach((card) => {
    card.addEventListener('click', () => {
      handlers.onChangeRosterFilter({ inspectedKey: card.dataset.inspectChar });
    });
  });

  container.querySelectorAll('[data-close-drawer]').forEach((el) => {
    el.addEventListener('click', (e) => {
      if (e.target === el || el.tagName === 'BUTTON') {
        handlers.onChangeRosterFilter({ inspectedKey: null });
      }
    });
  });

  const form = container.querySelector('[data-inspector-form]');
  if (form && !isReadOnly) {
    form.addEventListener('submit', async (e) => {
      e.preventDefault();
      const saveBtn = form.querySelector('button[type="submit"]');
      const origText = saveBtn?.textContent || 'Save to Sheet';
      if (saveBtn) {
        saveBtn.disabled = true;
        saveBtn.innerHTML = '<span class="btn-spinner"></span> Saving...';
      }
      try {
        const fd = new FormData(form);
        const clamp = (val, min, max, fallback) => {
          const n = Number(val);
          return Number.isFinite(n) ? Math.min(max, Math.max(min, n)) : fallback;
        };
        const rankRaw = String(fd.get('priorityRank') || '').trim();
        await handlers.onSaveCharacterInspector(form.dataset.inspectorForm, {
          status: String(fd.get('status') || 'Bench'),
          priorityRank: rankRaw ? clamp(rankRaw, 1, 99, '') : '',
          level: clamp(fd.get('level'), 1, 100, 1),
          targetLevel: clamp(fd.get('targetLevel'), 1, 100, 90),
          talent: {
            auto: clamp(fd.get('tAuto'), 1, 10, 1),
            skill: clamp(fd.get('tSkill'), 1, 10, 1),
            burst: clamp(fd.get('tBurst'), 1, 10, 1)
          },
          notes: String(fd.get('notes') || '')
        });
      } finally {
        if (saveBtn) {
          saveBtn.disabled = false;
          saveBtn.textContent = origText;
        }
      }
    });
  }
}

/**
 * Google Apps Script — Session-Token Auth Helpers (auth-session.gs)
 *
 * Script Properties (Project Settings -> Script Properties):
 *   PASS_almondeluna   = <Luna's passphrase>
 *   PASS_vanbrohenheim = <Roy's passphrase>
 *
 * Shared global helpers called by Code.gs (no doGet/doPost here).
 */

const SESSION_TTL_SECONDS = 6 * 60 * 60; // 6 hours
const SESSION_CACHE_PREFIX = 'genshin_session_';

function getExpectedPassphrase_(user) {
  const props = PropertiesService.getScriptProperties();
  return props.getProperty('PASS_' + String(user || '').trim().toLowerCase()) || '';
}

function createSessionToken_(user) {
  const token = Utilities.getUuid();
  CacheService.getScriptCache().put(
    SESSION_CACHE_PREFIX + token,
    JSON.stringify({ user: user, createdAt: Date.now() }),
    SESSION_TTL_SECONDS
  );
  return token;
}

function requireSession_(token) {
  if (!token) {
    throw new Error('Unauthorized: missing session token');
  }
  const raw = CacheService.getScriptCache().get(SESSION_CACHE_PREFIX + token);
  if (!raw) {
    throw new Error('Unauthorized: session expired or invalid');
  }
  return JSON.parse(raw);
}

function revokeSession_(token) {
  if (token) {
    CacheService.getScriptCache().remove(SESSION_CACHE_PREFIX + token);
  }
}

const TABLE_SCHEMAS = {

function ensureUserSheet_(ss, user, table) {
  const tabName = `${user}_${table}`;
  const sheet = ss.getSheetByName(tabName) || ss.insertSheet(tabName);
  if (sheet.getLastRow() === 0 && TABLE_SCHEMAS[table]) {
    sheet.appendRow(TABLE_SCHEMAS[table]);
  }
  return sheet;
}

function setupDatabase() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const profiles = ['almondeluna', 'vanbrohenheim'];

  // 1. Users Tab (display metadata; passphrases live in Script Properties)

function loadProfileAccount_(ss, targetUser) {
  const users = sheetToObjects(ss.getSheetByName('Users'));
  const profileMeta = users.find(u => String(u.username).toLowerCase() === targetUser) || {};

  const invRows = sheetToObjects(ss.getSheetByName(`${targetUser}_Inventory`));
  const inventory = {};
  invRows.forEach(r => {
    if (r.item_key) {
      inventory[r.item_key] = { count: Number(r.raw_count) || 0 };
    }
  });

  const sectionRows = sheetToObjects(ss.getSheetByName(`${targetUser}_Sections`));
  const sections = {};
  sectionRows.forEach(r => {
    if (r.section_key && r.json_value) {
      try {
        sections[r.section_key] = JSON.parse(r.json_value);
      } catch (err) {
        // Ignore malformed section JSON
      }
    }
  });

  return {
    user: targetUser,
    displayName: profileMeta.display_name || (targetUser === 'almondeluna' ? 'Luna' : 'Roy'),
    uid: profileMeta.uid || '',
    inventory: inventory,
    roster: sheetToObjects(ss.getSheetByName(`${targetUser}_Roster`)),
    weapons: sheetToObjects(ss.getSheetByName(`${targetUser}_Weapons`)),
    teams: sections.teams || sheetToObjects(ss.getSheetByName(`${targetUser}_Teams`)),
    tasks: sheetToObjects(ss.getSheetByName(`${targetUser}_Tasks`)),
    wallet: sections.wallet || null,
    checklist: sections.checklist || null,
    prefarmQueue: sections.prefarmQueue || null,
    characters: sections.characters || null
  };
}

function doGet(e) {
  try {
    const params = (e && e.parameter) || {};
    const session = requireSession_(params.token);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const action = params.action || 'getSheet';

    if (action === 'getSheet' || action === 'loadProfile') {
      const targetUser = String(params.user || params.profile || session.user).trim().toLowerCase();
      return jsonResponse({
        ok: true,
        profile: targetUser,
        account: loadProfileAccount_(ss, targetUser)
      });
    }

    return jsonResponse({ ok: false, error: 'Unknown action: ' + action });
  } catch (err) {
    return jsonResponse({ ok: false, error: err.message || String(err) });
  }
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);

  try {
    const payload = JSON.parse((e && e.postData && e.postData.contents) || '{}');
    const action = payload.action || '';
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const now = new Date().toISOString();

    if (action === 'login') {
      const user = String(payload.user || payload.username || '').trim().toLowerCase();
      const passphrase = String(payload.passphrase || payload.password || '');
      const expected = getExpectedPassphrase_(user);

      if (!expected || passphrase !== expected) {
        return jsonResponse({ ok: false, error: 'Invalid credentials' });
      }

      const sessionToken = createSessionToken_(user);
      return jsonResponse({
        ok: true,
        sessionToken: sessionToken,
        account: loadProfileAccount_(ss, user)
      });
    }

    if (action === 'logout') {
      revokeSession_(payload.token);
      return jsonResponse({ ok: true });
    }

    // All write actions require a valid session token matching the target user
    const session = requireSession_(payload.token);
    const user = String(payload.user || payload.profile || '').trim().toLowerCase();
    if (!user || session.user !== user) {
      return jsonResponse({ ok: false, error: 'Forbidden: cannot modify another user\'s sheet' });
    }

    // ACTION 1: Batch Import Irminsul GOOD v3 JSON
    if (action === 'importGoodJson') {
      const good = payload.good || payload.data || {};

      if (good.materials) {
        const invSheet = ensureUserSheet_(ss, user, 'Inventory');
        const existingInv = sheetToObjects(invSheet);
        const invMap = {};
        existingInv.forEach(r => { invMap[r.item_key] = r.raw_count; });
        Object.keys(good.materials).forEach(k => { invMap[k] = good.materials[k]; });

        const invRows = Object.keys(invMap).map(k => [k, invMap[k], now]);
        replaceSheetBody(invSheet, invRows);
      }

      if (Array.isArray(good.characters)) {
        const rosSheet = ensureUserSheet_(ss, user, 'Roster');
        const existingRos = sheetToObjects(rosSheet);
        const metaMap = {};
        existingRos.forEach(r => { metaMap[r.char_key] = r; });

        const rosRows = good.characters.map(c => {
          const prev = metaMap[c.key] || {};
          const autoStatus = c.level >= 90 ? 'Built' : (c.level >= 70 ? 'In Progress' : 'Bench');
          return [
            c.key,
            true,
            c.level,
            c.constellation,
            c.ascension,
            c.talent?.auto || 1,
            c.talent?.skill || 1,
            c.talent?.burst || 1,
            prev.target_lvl || 90,
            prev.target_talents || '9/9/9',
            prev.status || autoStatus,
            prev.priority_rank || '',
            prev.notes || ''
          ];
        });
        replaceSheetBody(rosSheet, rosRows);
      }

      if (Array.isArray(good.weapons)) {
        const wepSheet = ensureUserSheet_(ss, user, 'Weapons');
        const wepRows = good.weapons.map((w, idx) => [
          `${w.key}_${idx + 1}`,
          w.key,
          w.level,
          w.ascension,
          w.refinement,
          w.location || '',
          90,
          w.level >= 90 ? 'Maxed' : 'Bench'
        ]);
        replaceSheetBody(wepSheet, wepRows);
      }

      return jsonResponse({ ok: true, importedAt: now });
    }

    // ACTION 2: Single Row Update (Live UI Edits)
    if (action === 'upsertRow') {
      const sheetName = payload.sheet || `${user}_${payload.table || 'Inventory'}`;
      const inferredTable = payload.table || (sheetName.startsWith(`${user}_`) ? sheetName.slice(user.length + 1) : '');
      const sheet = ss.getSheetByName(sheetName) || (TABLE_SCHEMAS[inferredTable] ? ensureUserSheet_(ss, user, inferredTable) : null);
      if (!sheet) {
        return jsonResponse({ ok: false, error: 'Sheet not found: ' + sheetName });
      }

      const fields = payload.fields || payload.data || {};
      const normalizedFields = { ...fields };
      if (normalizedFields.raw_count === undefined && normalizedFields.count !== undefined) {
        normalizedFields.raw_count = normalizedFields.count;
      }
      if (normalizedFields.last_updated === undefined) {
        normalizedFields.last_updated = now;
      }

      const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
      const data = sheet.getDataRange().getValues();
      let targetRow = -1;

      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]) === String(payload.key)) {
          targetRow = i + 1;
          break;
        }
      }

      if (targetRow === -1) {
        const newRow = headers.map((h, idx) => {
          if (idx === 0) return payload.key;
          return normalizedFields[h] !== undefined ? normalizedFields[h] : '';
        });
        sheet.appendRow(newRow);
      } else {
        Object.keys(normalizedFields).forEach(colName => {
          const colIdx = headers.indexOf(colName);
          if (colIdx !== -1) {
            sheet.getRange(targetRow, colIdx + 1).setValue(normalizedFields[colName]);
          }
        });
      }
      return jsonResponse({ ok: true });
    }

    // ACTION 3: Upsert JSON Section (wallet, checklist, prefarmQueue, characters, teams)
    if (action === 'upsertSection') {
      const secSheet = ss.getSheetByName(`${user}_Sections`) || ss.insertSheet(`${user}_Sections`);
      if (secSheet.getLastRow() === 0) {
        secSheet.appendRow(['section_key', 'json_value', 'last_updated']);
      }
      const data = secSheet.getDataRange().getValues();
      let targetRow = -1;
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]) === String(payload.section)) {
          targetRow = i + 1;
          break;
        }
      }
      const serialized = JSON.stringify(payload.data);
      if (targetRow === -1) {
        secSheet.appendRow([payload.section, serialized, now]);
      } else {
        secSheet.getRange(targetRow, 2, 1, 2).setValues([[serialized, now]]);
      }
      return jsonResponse({ ok: true });
    }

    return jsonResponse({ ok: false, error: 'Unknown POST action: ' + action });
  } catch (err) {
    return jsonResponse({ ok: false, error: err.message || String(err) });
  } finally {
    lock.releaseLock();
  }
}

function sheetToObjects(sheet) {
  if (!sheet || sheet.getLastRow() < 2) return [];
  const values = sheet.getDataRange().getValues();
  const headers = values[0];
  return values.slice(1).map(row => {
    const obj = {};
    headers.forEach((h, i) => { obj[h] = row[i]; });
    return obj;
  });
}

function replaceSheetBody(sheet, rows) {
  if (sheet.getLastRow() > 1) {
    sheet.getRange(2, 1, sheet.getLastRow() - 1, sheet.getLastColumn()).clearContent();
  }
  if (rows.length > 0) {
    sheet.getRange(2, 1, rows.length, rows[0].length).setValues(rows);
  }
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
import genshindb from 'genshin-db';

const __dirname = path.dirname(fileURLToPath(import.meta.url));
const ROOT_DIR = path.resolve(__dirname, '..');
const OUTPUT_PATH = path.join(ROOT_DIR, 'data', 'catalog.json');

/**
 * Converts a display name into a GOOD v3 PascalCase key.
 */
function toPascalKey(name) {
  if (!name) return '';
  return String(name)
    .replace(/['’`"]/g, '')
    .replace(/[^a-zA-Z0-9]+/g, ' ')
    .trim()
    .split(/\s+/)
    .map((word) => word.charAt(0).toUpperCase() + word.slice(1))
    .join('');
}

function resolveIconUrl(images) {
  if (!images) return '';
  return (
    images.mihoyo_icon ||
    images.icon ||
    images.filename_icon ||
    images.cover1 ||
    ''
  );
}

function buildCharacters() {
  const names = genshindb.characters('names', { matchCategories: true }) || [];
  const characters = {};

  for (const name of names) {
    const c = genshindb.characters(name);
    if (!c) continue;
    const key = toPascalKey(c.name);
    const talentData = genshindb.talents(name);

    const ascensionMaterials = {};
    if (c.costs) {
      for (const [phase, items] of Object.entries(c.costs)) {
        ascensionMaterials[phase] = (items || []).map((item) => ({
          key: toPascalKey(item.name),
          name: item.name,
          count: item.count
        }));
      }
    }

    const talentMaterials = {};
    if (talentData?.costs) {
      for (const [lvl, items] of Object.entries(talentData.costs)) {
        talentMaterials[lvl] = (items || []).map((item) => ({
          key: toPascalKey(item.name),
          name: item.name,
          count: item.count
        }));
      }
    }

    characters[key] = {
      key,
      name: c.name,
      rarity: Number(c.rarity) || 4,
      element: c.elementText || 'None',
      weaponType: c.weaponText || 'Sword',
      region: c.region || 'Unknown',
      icon: resolveIconUrl(c.images),
      ascensionMaterials,
      talentMaterials
    };
  }

  return characters;
}

function buildWeapons() {
  const names = genshindb.weapons('names', { matchCategories: true }) || [];
  const weapons = {};

  for (const name of names) {
    const w = genshindb.weapons(name);
    if (!w) continue;
    const key = toPascalKey(w.name);
    const ascensionMaterials = {};

    if (w.costs) {
      for (const [phase, items] of Object.entries(w.costs)) {
        ascensionMaterials[phase] = (items || []).map((item) => ({
          key: toPascalKey(item.name),
          name: item.name,
          count: item.count
        }));
      }
    }

    weapons[key] = {
      key,
      name: w.name,
      rarity: Number(w.rarity) || 3,
      weaponType: w.weaponText || 'Sword',
      substat: w.mainStatText || '',
      icon: resolveIconUrl(w.images),
      ascensionMaterials
    };
  }

  return weapons;
}

function buildMaterials() {
  const names = genshindb.materials('names', { matchCategories: true }) || [];
  const materials = {};

  for (const name of names) {
    const m = genshindb.materials(name);
    if (!m) continue;
    const key = toPascalKey(m.name);

    materials[key] = {
      key,
      name: m.name,
      rarity: Number(m.rarity) || 1,
      category: m.category || m.typeText || 'Material',
      daysOfWeek: m.daysOfWeek || [],
      dropDomain: m.dropDomainName || '',
      icon: resolveIconUrl(m.images)
    };
  }

  return materials;
}

function buildDomainSchedule() {
  const domains = genshindb.domains('names', { matchCategories: true }) || [];
  const schedule = {
    Monday: [],
    Tuesday: [],
    Wednesday: [],
    Thursday: [],
    Friday: [],
    Saturday: [],
    Sunday: []
  };

  for (const name of domains) {
    const d = genshindb.domains(name);
    if (!d || !Array.isArray(d.daysOfWeek)) continue;
    for (const day of d.daysOfWeek) {
      if (!schedule[day]) continue;
      schedule[day].push({
        name: d.name,
        region: d.regionName || '',
        domainType: d.domainType || '',
        rewards: (d.rewardPreview || []).map((r) => ({
          key: toPascalKey(r.name),
          name: r.name
        }))
      });
    }
  }

  return schedule;
}

==================== sync-snapshot.mjs ====================

/**
 * Polite snapshot sync script for GitHub Actions.
 * Checks api.lunaris.moe/data/version.json first and exits early if unchanged.
 * Compiles data/live-snapshot.json from Project Amber (gi.yatta.moe) + Lunaris.
 */
import fs from 'node:fs';
import path from 'node:path';

const ROOT = process.cwd();
const CATALOG_PATH = path.join(ROOT, 'data', 'catalog.json');
const SNAPSHOT_PATH = path.join(ROOT, 'data', 'live-snapshot.json');
const YATTA_UI_BASE = 'https://gi.yatta.moe/assets/UI/';
const HEADERS = { 'User-Agent': 'aaru-db-snapshot-sync/1.0' };

const ELEMENT_MAP = {
  Fire: 'Pyro',
  Water: 'Hydro',
  Wind: 'Anemo',
  Electric: 'Electro',
  Grass: 'Dendro',
  Ice: 'Cryo',
  Rock: 'Geo'
};

const WEAPON_TYPE_MAP = {
  WEAPON_SWORD_ONE_HAND: 'Sword',
  WEAPON_CLAYMORE: 'Claymore',
  WEAPON_POLE: 'Polearm',
  WEAPON_CATALYST: 'Catalyst',
  WEAPON_BOW: 'Bow'
};

function normalizeAssetKey(str) {
  return String(str || '').toLowerCase().replace(/[^a-z0-9]/g, '');
}

function toPascal(str) {
  return String(str || '').replace(/[^a-zA-Z0-9]/g, '');
}

async function fetchJson(url) {
  const res = await fetch(url, { headers: HEADERS });
  if (!res.ok) throw new Error(`${url} -> HTTP ${res.status}`);
  return res.json();
}

const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

async function main() {
  const force = process.argv.includes('--force');
  const verData = await fetchJson('https://api.lunaris.moe/data/version.json');
  const lunarisVer = verData?.version;
  if (!lunarisVer) throw new Error('Missing Lunaris version');

  let existing = null;
  if (fs.existsSync(SNAPSHOT_PATH)) {
    try {
      existing = JSON.parse(fs.readFileSync(SNAPSHOT_PATH, 'utf8'));
    } catch {
      existing = null;
    }
  }

  if (!force && existing?._lunarisVer === lunarisVer) {
    console.log(`Snapshot already up to date (${lunarisVer}).`);
    return;
  }

  console.log(`Syncing snapshot for Lunaris ${lunarisVer} (previous: ${existing?._lunarisVer || 'none'})...`);

  const baseCatalog = JSON.parse(fs.readFileSync(CATALOG_PATH, 'utf8'));
  const domainRewardNormSet = new Set();
  for (const dayList of Object.values(baseCatalog?.domainSchedule || {})) {
    for (const dom of dayList || []) {
      for (const r of dom.rewards || []) domainRewardNormSet.add(normalizeAssetKey(r.key));
    }
  }

  const [avatarRes, weaponRes, materialRes, upgradeRes, monsterRes] = await Promise.allSettled([
    fetchJson('https://gi.yatta.moe/api/v2/en/avatar'),
    fetchJson('https://gi.yatta.moe/api/v2/en/weapon'),
    fetchJson('https://gi.yatta.moe/api/v2/en/material'),
    fetchJson('https://gi.yatta.moe/api/v2/en/upgrade'),
    fetchJson('https://gi.yatta.moe/api/v2/en/monster')
  ]);

  const slim = {
    _lunarisVer: lunarisVer,
    _syncedAt: new Date().toISOString(),
    _matIdLookup: {},
    characters: {},
    weapons: {},
    materials: {}
  };

  const matNormById = new Map();
  if (materialRes.status === 'fulfilled' && materialRes.value?.data?.items) {
    for (const [id, item] of Object.entries(materialRes.value.data.items)) {
      const norm = normalizeAssetKey(item.name);
      if (norm) matNormById.set(String(id), { norm, name: item.name, rank: Number(item.rank) || 2 });
      if (!norm || !item.icon) continue;
      slim.materials[norm] = {
        name: item.name,
        rarity: Number(item.rank) || 2,
        category: item.type || 'Material',
        icon: `${YATTA_UI_BASE}${item.icon}.png`
      };
    }
  }

  const avatarUpgradeByNorm = new Map();
  const weaponUpgradeByNorm = new Map();
  if (upgradeRes.status === 'fulfilled' && upgradeRes.value?.data) {
    for (const item of Object.values(upgradeRes.value.data.avatar || {})) {
      const norm = normalizeAssetKey(item.name);
      if (!norm || !item.items) continue;
      const upg = {};
      for (const matId of Object.keys(item.items)) {
        const mat = matNormById.get(String(matId));
        if (!mat) continue;
        if (!upg.talentBook && (mat.name.startsWith('Philosophies of ') || domainRewardNormSet.has(mat.norm))) {
          upg.domainMatKey = mat.norm;
          upg.talentBook = mat.name.replace(/^Philosophies of\s+/i, '');
        } else if (/^113\d+$/.test(String(matId))) {
          const pascalMat = toPascal(mat.name);
          if (mat.rank === 4 && !upg.bossMat) upg.bossMat = pascalMat;
          else if (mat.rank === 5 && !upg.weeklyMat) upg.weeklyMat = pascalMat;
        }
      }
      if (Object.keys(upg).length) avatarUpgradeByNorm.set(norm, upg);
    }

    for (const item of Object.values(upgradeRes.value.data.weapon || {})) {
      const norm = normalizeAssetKey(item.name);
      if (!norm || !item.items) continue;
      for (const matId of Object.keys(item.items)) {
        const mat = matNormById.get(String(matId));
        if (mat && domainRewardNormSet.has(mat.norm)) {
          weaponUpgradeByNorm.set(norm, {
            domainMatKey: mat.norm,
            domainMatExactKey: toPascal(mat.name)
          });
          break;
        }
      }
    }
  }

  if (avatarRes.status === 'fulfilled' && avatarRes.value?.data?.items) {
    for (const item of Object.values(avatarRes.value.data.items)) {
      const norm = normalizeAssetKey(item.name);
      if (!norm || !item.icon) continue;
      const upg = avatarUpgradeByNorm.get(norm);
      slim.characters[norm] = {
        pascalKey: toPascal(item.name),
        name: item.name,
        rarity: Number(item.rank) || 5,
        element: ELEMENT_MAP[item.element] || item.element || 'Unknown',
        weaponType: WEAPON_TYPE_MAP[item.weaponType] || 'Sword',
        region: item.region || 'Teyvat',
        icon: `${YATTA_UI_BASE}${item.icon}.png`,
        ...(upg || {})
      };
    }
  }

  if (weaponRes.status === 'fulfilled' && weaponRes.value?.data?.items) {
    for (const item of Object.values(weaponRes.value.data.items)) {
      const norm = normalizeAssetKey(item.name);
      if (!norm || !item.icon || item.isWeaponSkin) continue;
      const upg = weaponUpgradeByNorm.get(norm);
      slim.weapons[norm] = {
        pascalKey: toPascal(item.name),
        name: item.name,
        rarity: Number(item.rank) || 4,
        weaponType: WEAPON_TYPE_MAP[item.type] || 'Sword',
        icon: `${YATTA_UI_BASE}${item.icon}.png`,
        ...(upg || {})
      };
    }
  }

  const [lCharRes, lWepRes, lMatRes, lMonRes] = await Promise.allSettled([
    fetchJson(`https://api.lunaris.moe/data/${encodeURIComponent(lunarisVer)}/charlist.json`),
    fetchJson(`https://api.lunaris.moe/data/${encodeURIComponent(lunarisVer)}/weaponlist.json`),
    fetchJson(`https://api.lunaris.moe/data/${encodeURIComponent(lunarisVer)}/materiallist.json`),
    fetchJson(`https://api.lunaris.moe/data/${encodeURIComponent(lunarisVer)}/monsterlist.json`)
  ]);

  const yattaMonsterIds = new Set(
    monsterRes.status === 'fulfilled' && monsterRes.value?.data?.items
      ? Object.keys(monsterRes.value.data.items)
      : []
  );
  const lunarisMonsters = lMonRes.status === 'fulfilled' && lMonRes.value ? lMonRes.value : {};

  for (const [id, m] of matNormById.entries()) {
    slim._matIdLookup[id] = m.name;
  }
  if (lMatRes.status === 'fulfilled' && lMatRes.value) {
    for (const [id, item] of Object.entries(lMatRes.value)) {
      const name = item?.enName;
      const norm = normalizeAssetKey(name);
      if (!norm) continue;
      slim._matIdLookup[String(id)] = name;
      if (!slim.materials[norm] && item.icon) {
        slim.materials[norm] = {
          name,
          rarity: item.qualityType === 'QUALITY_ORANGE' ? 5 : item.qualityType === 'QUALITY_PURPLE' ? 4 : 3,
          category: item.type || 'Material',
          icon: `https://api.lunaris.moe/data/assets/items/${item.icon}.webp`
        };
      }
      if (/^113\d+$/.test(String(id)) && Array.isArray(item.sourceMonsters) && slim.materials[norm]?.rarity === 4) {
        const monId = String(item.sourceMonsters?.[0]?.[0] || '');
        const mon = monId ? lunarisMonsters[monId] : null;
        if (mon?.enName) {
          slim.materials[norm].bossName = mon.enName;
          slim.materials[norm].bossIcon = mon.icon
            ? `https://api.lunaris.moe/data/assets/monstericon/${mon.icon}.webp`
            : '';
          slim.materials[norm].bossUnreleased = yattaMonsterIds.size > 0 && !yattaMonsterIds.has(monId);
        }
      }
    }
  }

  // Merge & hydrate Lunaris-only characters gently
  if (lCharRes.status === 'fulfilled' && lCharRes.value) {
    for (const [charId, item] of Object.entries(lCharRes.value)) {
      const name = item?.enName;
      const norm = normalizeAssetKey(name);
      if (!norm || norm === 'traveler' || slim.characters[norm]) continue;
      const iconName = String(item.CardImg || '').replace(/^UI_Gacha_AvatarIcon_/, 'UI_AvatarIcon_');
      const entry = {
        lunarisId: charId,
        pascalKey: toPascal(name),
        name,
        rarity: item.qualityType === 'QUALITY_ORANGE' ? 5 : 4,
        element: item.element || 'Unknown',
        weaponType: WEAPON_TYPE_MAP[item.weaponType] || 'Sword',
        region: 'Teyvat',
        icon: iconName ? `https://api.lunaris.moe/data/assets/avataricon/${iconName}.webp` : ''
      };
      try {
        await sleep(100);
        const detail = await fetchJson(`https://api.lunaris.moe/data/${encodeURIComponent(lunarisVer)}/en/char/${encodeURIComponent(charId)}.json`);
        for (const matId of Object.keys(detail?.skills?.leveling?.books || {})) {
          const matName = slim._matIdLookup[String(matId)];
          if (matName && /^Philosophies of /i.test(matName)) {
            entry.talentBook = matName.replace(/^Philosophies of\s+/i, '');
            entry.domainMatKey = normalizeAssetKey(matName);
            break;
          }
        }
        const asc = Array.isArray(detail?.ascension) ? detail.ascension[detail.ascension.length - 1] : detail?.ascension || null;
        const bossId = asc?.elitemonster ? Object.keys(asc.elitemonster)[0] : null;
        if (bossId && slim._matIdLookup[bossId]) entry.bossMat = toPascal(slim._matIdLookup[bossId]);
        const specId = asc?.speciality ? Object.keys(asc.speciality)[0] : null;
        if (specId && slim._matIdLookup[specId]) entry.localSpecialty = toPascal(slim._matIdLookup[specId]);
        const weeklyId = Object.keys(detail?.skills?.leveling?.weekly || {})[0];
        if (weeklyId && slim._matIdLookup[weeklyId]) entry.weeklyMat = toPascal(slim._matIdLookup[weeklyId]);
        entry._hydrated = true;
      } catch {
        // Ignore individual character hydration failures
      }
      slim.characters[norm] = entry;
    }
  }

  // Merge & hydrate Lunaris-only weapons gently
  if (lWepRes.status === 'fulfilled' && lWepRes.value) {
    for (const [wepId, item] of Object.entries(lWepRes.value)) {
      const name = item?.enName;
      const norm = normalizeAssetKey(name);
      if (!norm || slim.weapons[norm]) continue;
      const entry = {
        lunarisId: wepId,
        pascalKey: toPascal(name),
        name,
        rarity: item.qualityType === 'QUALITY_ORANGE' ? 5 : item.qualityType === 'QUALITY_PURPLE' ? 4 : 3,
        weaponType: WEAPON_TYPE_MAP[item.weaponType] || 'Sword',
        icon: item.weaponIcon ? `https://api.lunaris.moe/data/assets/weaponicon/${item.weaponIcon}.webp` : ''
      };
      try {
        await sleep(100);
        const detail = await fetchJson(`https://api.lunaris.moe/data/${encodeURIComponent(lunarisVer)}/en/weapon/${encodeURIComponent(wepId)}.json`);
        const asc = Array.isArray(detail?.ascension) ? detail.ascension : [];
        const domainMats = asc.map((a) => String(a?.icon || '').replace(/^UI_ItemIcon_/, '')).filter((id) => /^114\d+$/.test(id));
        const goldId = domainMats[domainMats.length - 1];
        if (goldId && slim._matIdLookup[goldId]) {
          entry.domainMatExactKey = toPascal(slim._matIdLookup[goldId]);
          entry.domainMatKey = normalizeAssetKey(slim._matIdLookup[goldId]);
        }
        entry._hydrated = true;
      } catch {
        // Ignore individual weapon hydration failures
      }
      slim.weapons[norm] = entry;
    }
  }

  fs.writeFileSync(SNAPSHOT_PATH, JSON.stringify(slim));
  console.log(`Wrote ${SNAPSHOT_PATH} (${Object.keys(slim.characters).length} chars, ${Object.keys(slim.weapons).length} weapons, ${Object.keys(slim.materials).length} mats).`);
}

main().catch((err) => {
  console.error(err);
  process.exit(1);
});
